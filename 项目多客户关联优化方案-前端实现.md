// src/api/evs/projectCustomers.js
import request from '@/utils/request'

// 查询项目的所有客户
export function getProjectCustomers(projectId) {
  return request({
    url: `/evs/project-customers/project/${projectId}`,
    method: 'get'
  })
}

// 查询客户的所有项目
export function getCustomerProjects(customerId) {
  return request({
    url: `/evs/project-customers/customer/${customerId}`,
    method: 'get'
  })
}

// 添加项目客户关联
export function addProjectCustomers(data) {
  return request({
    url: '/evs/project-customers',
    method: 'post',
    data: data
  })
}

// 移除项目客户关联
export function removeProjectCustomer(projectId, customerId) {
  return request({
    url: `/evs/project-customers/${projectId}/${customerId}`,
    method: 'delete'
  })
}

// 设置主客户
export function setPrimaryCustomer(projectId, customerId) {
  return request({
    url: `/evs/project-customers/primary/${projectId}/${customerId}`,
    method: 'put'
  })
}

// 更新客户角色
export function updateCustomerRole(data) {
  return request({
    url: '/evs/project-customers/role',
    method: 'put',
    data: data
  })
}

<template>
  <el-dialog
    :title="title"
    v-model="dialogVisible"
    width="800px"
    :close-on-click-modal="false"
    @close="handleClose"
  >
    <el-form ref="formRef" :model="form" :rules="rules" label-width="100px">

        <el-input v-model="form.name" placeholder="请输入项目名称" />

            filterable

              <el-select

              >
                <el-option label="业主" value="OWNER" />
                <el-option label="共同业主" value="CO_OWNER" />
                <el-option label="投资方" value="INVESTOR" />

<script setup>
import { ref, reactive, computed, watch } from 'vue'
import { ElMessage } from 'element-plus'
import { listCustomers } from '@/api/evs/customers'
import { addProject, updateProject } from '@/api/evs/projects'

const emit = defineEmits(['success'])

// 对话框状态
const dialogVisible = ref(false)
const title = ref('新增项目')
const submitLoading = ref(false)

// 表单数据
const form = reactive({
  id: null,
  name: '',
  customerIds: [],
  primaryCustomerId: null,
  address: '',
  area: null,
  projectCustomers: [] // 客户详细信息
})

// 客户列表
const customerList = ref([])

// 表单验证规则
const rules = {
  name: [
    { required: true, message: '请输入项目名称', trigger: 'blur' }
  ],
  customerIds: [
    { required: true, message: '请至少选择一个客户', trigger: 'change' },
    { 
      validator: (rule, value, callback) => {
        if (value.length > 10) {
          callback(new Error('最多只能选择10个客户'))
        } else {
          callback()
        }
      },
      trigger: 'change'
    }
  ],
  primaryCustomerId: [
    { required: true, message: '请选择主客户', trigger: 'change' }
  ]
}

// 已选客户的详细数据
const selectedCustomersData = computed(() => {
  return form.customerIds.map(customerId => {
    const customer = customerList.value.find(c => c.id === customerId)
    const existingData = form.projectCustomers.find(pc => pc.customerId === customerId)
    
    return {
      customerId: customerId,
      name: customer?.name || '',
      phone: customer?.phone || '',
      role: existingData?.role || 'OWNER',
      shareRatio: existingData?.shareRatio || 0,
      isPrimary: customerId === form.primaryCustomerId
    }
  })
})

// 监听客户选择变化
watch(() => form.customerIds, (newVal) => {
  // 如果只有一个客户，自动设为主客户
  if (newVal.length === 1) {
    form.primaryCustomerId = newVal[0]
  }
  // 如果主客户被移除，清空主客户选择
  if (form.primaryCustomerId && !newVal.includes(form.primaryCustomerId)) {
    form.primaryCustomerId = null
  }
})

// 获取客户名称
function getCustomerName(customerId) {
  const customer = customerList.value.find(c => c.id === customerId)
  return customer ? `${customer.name} (${customer.phone})` : customerId
}

// 移除客户
function removeCustomer(customerId) {
  const index = form.customerIds.indexOf(customerId)
  if (index > -1) {
    form.customerIds.splice(index, 1)
  }
}

// 客户选择变化
function handleCustomerChange(value) {
  // 限制最多10个
  if (value.length > 10) {
    ElMessage.warning('最多只能选择10个客户')
    form.customerIds = value.slice(0, 10)
  }
}

// 打开对话框
async function show(row = null) {
  dialogVisible.value = true
  title.value = row ? '编辑项目' : '新增项目'
  
  // 加载客户列表
  await loadCustomers()
  
  if (row) {
    // 编辑模式
    Object.assign(form, row)
    // 如果有客户关联数据，加载
    if (row.projectCustomers && row.projectCustomers.length > 0) {
      form.customerIds = row.projectCustomers.map(pc => pc.customerId)
      form.projectCustomers = row.projectCustomers
      
      // 找出主客户
      const primary = row.projectCustomers.find(pc => pc.isPrimary)
      form.primaryCustomerId = primary ? primary.customerId : row.customerId
    } else if (row.customerId) {
      // 兼容旧数据
      form.customerIds = [row.customerId]
      form.primaryCustomerId = row.customerId
    }
  } else {
    // 新增模式
    resetForm()
  }
}

// 加载客户列表
async function loadCustomers() {
  try {
    const response = await listCustomers({})
    customerList.value = response.rows || []
  } catch (error) {
    ElMessage.error('加载客户列表失败')
  }
}

// 提交表单
async function handleSubmit() {
  // 验证表单
  const valid = await formRef.value.validate().catch(() => false)
  if (!valid) return
  
  submitLoading.value = true
  
  try {
    // 构建客户关联数据
    const projectCustomers = selectedCustomersData.value.map(item => ({
      customerId: item.customerId,
      role: item.role,
      shareRatio: item.shareRatio,
      isPrimary: item.customerId === form.primaryCustomerId
    }))
    
    const data = {
      ...form,
      customerId: form.primaryCustomerId, // 主客户ID
      projectCustomers: projectCustomers
    }
    
    if (form.id) {
      await updateProject(data)
      ElMessage.success('修改成功')
    } else {
      await addProject(data)
      ElMessage.success('新增成功')
    }
    
    emit('success')
    handleClose()
  } catch (error) {
    ElMessage.error(error.message || '操作失败')
  } finally {
    submitLoading.value = false
  }
}

// 重置表单
function resetForm() {
  form.id = null
  form.name = ''
  form.customerIds = []
  form.primaryCustomerId = null
  form.address = ''
  form.area = null
  form.projectCustomers = []
}

// 关闭对话框
function handleClose() {
  dialogVisible.value = false
  resetForm()
}

defineExpose({ show })
</script>

  color: #909399;

}

<script setup>
import { ref, computed, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import {
  getProjectCustomers,
  addProjectCustomers,
  removeProjectCustomer,
  setPrimaryCustomer,
  updateCustomerRole
} from '@/api/evs/projectCustomers'
import { listCustomers } from '@/api/evs/customers'

const props = defineProps({
  projectId: {
    type: String,
    required: true
  },
  canEdit: {
    type: Boolean,
    default: true
  }
})

// 客户列表
const customerList = ref([])
const allCustomers = ref([])

// 添加对话框
const addDialogVisible = ref(false)
const addForm = ref({
  customerIds: [],
  primaryCustomerId: null
})

// 编辑对话框
const editDialogVisible = ref(false)
const editForm = ref({
  id: null,
  customerName: '',
  role: '',
  shareRatio: 0
})

// 可选客户列表（排除已关联的）
const availableCustomers = computed(() => {
  const existingIds = customerList.value.map(c => c.customerId)
  return allCustomers.value.filter(c => !existingIds.includes(c.id))
})

onMounted(() => {
  loadData()
  loadAllCustomers()
})

// 加载数据
async function loadData() {
  try {
    const response = await getProjectCustomers(props.projectId)
    customerList.value = response.data || []
  } catch (error) {
    ElMessage.error('加载客户列表失败')
  }
}

// 加载所有客户
async function loadAllCustomers() {
  try {
    const response = await listCustomers({})
    allCustomers.value = response.rows || []
  } catch (error) {
    ElMessage.error('加载客户列表失败')
  }
}

// 获取角色文本
function getRoleText(role) {
  const roleMap = {
    'OWNER': '业主',
    'CO_OWNER': '共同业主',
    'INVESTOR': '投资方',
    'FAMILY': '家庭成员'
  }
  return roleMap[role] || role
}

// 获取角色标签类型
function getRoleType(role) {
  const typeMap = {
    'OWNER': 'success',
    'CO_OWNER': 'primary',
    'INVESTOR': 'warning',
    'FAMILY': 'info'
  }
  return typeMap[role] || ''
}

// 格式化日期
function formatDate(date) {
  if (!date) return '-'
  return new Date(date).toLocaleString('zh-CN')
}

// 根据ID获取客户名称
function getCustomerNameById(customerId) {
  const customer = allCustomers.value.find(c => c.id === customerId)
  return customer ? customer.name : customerId
}

// 添加客户
function handleAdd() {
  if (customerList.value.length >= 10) {
    ElMessage.warning('项目最多只能关联10个客户')
    return
  }
  addForm.value = {
    customerIds: [],
    primaryCustomerId: null
  }
  addDialogVisible.value = true
}

// 提交添加
async function handleAddSubmit() {
  if (addForm.value.customerIds.length === 0) {
    ElMessage.warning('请选择至少一个客户')
    return
  }
  
  try {
    await addProjectCustomers({
      projectId: props.projectId,
      customerIds: addForm.value.customerIds,
      primaryCustomerId: addForm.value.primaryCustomerId
    })
    ElMessage.success('添加成功')
    addDialogVisible.value = false
    loadData()
  } catch (error) {
    ElMessage.error(error.message || '添加失败')
  }
}

// 设为主客户
async function handleSetPrimary(row) {
  try {
    await ElMessageBox.confirm(
      `确定将"${row.customer.name}"设为主客户吗？`,
      '提示',
      { type: 'warning' }
    )
    
    await setPrimaryCustomer(props.projectId, row.customerId)
    ElMessage.success('设置成功')
    loadData()
  } catch (error) {
    if (error !== 'cancel') {
      ElMessage.error(error.message || '设置失败')
    }
  }
}

// 编辑客户
function handleEdit(row) {
  editForm.value = {
    id: row.id,
    customerName: row.customer.name,
    role: row.role,
    shareRatio: row.shareRatio || 0
  }
  editDialogVisible.value = true
}

// 提交编辑
async function handleEditSubmit() {
  try {
    await updateCustomerRole(editForm.value)
    ElMessage.success('更新成功')
    editDialogVisible.value = false
    loadData()
  } catch (error) {
    ElMessage.error(error.message || '更新失败')
  }
}

// 移除客户
async function handleRemove(row) {
  try {
    await ElMessageBox.confirm(
      `确定移除客户"${row.customer.name}"吗？`,
      '提示',
      { type: 'warning' }
    )
    
    await removeProjectCustomer(props.projectId, row.customerId)
    ElMessage.success('移除成功')
    loadData()
  } catch (error) {
    if (error !== 'cancel') {
      ElMessage.error(error.message || '移除失败')
    }
  }
}
</script>
