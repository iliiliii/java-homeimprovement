# 项目多客户关联优化方案

## 一、需求分析

### 1.1 当前架构
```
projects 表
├─ customer_id (VARCHAR(32))  # 单一客户ID
└─ 关系: 一对一（一个项目 → 一个客户）
```

### 1.2 新需求
```
一个项目可以关联多个客户（最多10个）
- 支持主客户和次客户的概念
- 需要记录客户在项目中的角色
- 需要支持客户的加入/退出时间
- 权限控制需要调整
```

### 1.3 业务场景
1. **合伙装修**: 多个业主共同装修一套房产
2. **家庭成员**: 父母、子女共同参与装修决策
3. **商业项目**: 多个投资方共同参与的商业装修
4. **分期项目**: 不同阶段由不同客户负责

## 二、数据库设计方案

### 方案A: 中间表方案（推荐）⭐

#### 2.1 创建项目客户关联表

```sql
-- 项目客户关联表
CREATE TABLE `project_customers` (
  `id` VARCHAR(32) NOT NULL COMMENT '关联ID',
  `project_id` VARCHAR(32) NOT NULL COMMENT '项目ID',
  `customer_id` VARCHAR(32) NOT NULL COMMENT '客户ID',
  `role` VARCHAR(20) NOT NULL DEFAULT 'OWNER' COMMENT '客户角色（OWNER:主客户、CO_OWNER:共同客户、INVESTOR:投资方、FAMILY:家庭成员）',
  `is_primary` TINYINT(1) NOT NULL DEFAULT 0 COMMENT '是否主客户（1:是 0:否）',
  `join_date` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '加入时间',
  `leave_date` DATETIME NULL COMMENT '退出时间',
  `is_active` TINYINT(1) NOT NULL DEFAULT 1 COMMENT '是否有效（1:有效 0:无效）',
  `share_ratio` DECIMAL(5,2) NULL COMMENT '份额比例（%）',
  `remarks` TEXT COMMENT '备注',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` DATETIME NULL ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `created_by` VARCHAR(64) NULL COMMENT '创建人',
  `updated_by` VARCHAR(64) NULL COMMENT '更新人',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_project_customer` (`project_id`, `customer_id`),
  KEY `idx_project_id` (`project_id`),
  KEY `idx_customer_id` (`customer_id`),
  KEY `idx_is_primary` (`is_primary`),
  KEY `idx_is_active` (`is_active`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='项目客户关联表';

-- 添加约束：每个项目只能有一个主客户
-- 通过应用层逻辑控制，或使用触发器
```

#### 2.2 修改 projects 表

```sql
-- 方式1: 保留 customer_id 作为主客户快捷字段（推荐）
ALTER TABLE `projects` 
  MODIFY COLUMN `customer_id` VARCHAR(32) NULL COMMENT '主客户ID（冗余字段，便于快速查询）';

-- 方式2: 废弃 customer_id，完全依赖关联表
ALTER TABLE `projects` 
  DROP COLUMN `customer_id`;
-- 注意：此方式需要修改大量现有代码
```

**推荐方式1的理由**:
- 保持向后兼容
- 快速查询主客户信息
- 减少代码改动量
- customer_id 作为冗余字段，与 project_customers 表同步维护

#### 2.3 数据迁移脚本

```sql
-- 将现有 customer_id 数据迁移到 project_customers 表
INSERT INTO `project_customers` 
  (`id`, `project_id`, `customer_id`, `role`, `is_primary`, `join_date`, `is_active`, `created_at`, `created_by`)
SELECT 
  UUID() as id,
  p.id as project_id,
  p.customer_id,
  'OWNER' as role,
  1 as is_primary,
  COALESCE(p.created_at, NOW()) as join_date,
  1 as is_active,
  NOW() as created_at,
  'system' as created_by
FROM projects p
WHERE p.customer_id IS NOT NULL 
  AND p.deleted_at IS NULL
  AND NOT EXISTS (
    SELECT 1 FROM project_customers pc 
    WHERE pc.project_id = p.id AND pc.customer_id = p.customer_id
  );
```

### 方案B: JSON 字段方案（不推荐）

```sql
-- 在 projects 表添加 JSON 字段
ALTER TABLE `projects` 
  ADD COLUMN `customer_ids` JSON COMMENT '客户ID列表（JSON数组）',
  ADD COLUMN `customer_details` JSON COMMENT '客户详细信息（JSON对象）';

-- 示例数据
{
  "customers": [
    {
      "customerId": "C001",
      "role": "OWNER",
      "isPrimary": true,
      "shareRatio": 60.0
    },
    {
      "customerId": "C002",
      "role": "CO_OWNER",
      "isPrimary": false,
      "shareRatio": 40.0
    }
  ]
}
```

**不推荐理由**:
- 查询性能差（无法有效使用索引）
- 数据完整性难以保证
- 关联查询复杂
- 不符合关系型数据库设计规范

### 3.3 创建 Mapper 接口

```java
package com.ruoyi.web.mapper;

import java.util.List;
import org.apache.ibatis.annotations.Mapper;
import org.apache.ibatis.annotations.Param;
import com.ruoyi.web.domain.ProjectCustomers;

@Mapper
public interface ProjectCustomersMapper {
    
    /**
     * 查询项目的所有客户
     */
    List<ProjectCustomers> selectByProjectId(@Param("projectId") String projectId);
    
    /**
     * 查询客户的所有项目
     */
    List<ProjectCustomers> selectByCustomerId(@Param("customerId") String customerId);
    
    /**
     * 查询项目的主客户
     */
    ProjectCustomers selectPrimaryByProjectId(@Param("projectId") String projectId);
    
    /**
     * 检查客户是否关联到项目
     */
    boolean checkCustomerInProject(@Param("projectId") String projectId, 
                                   @Param("customerId") String customerId);
    
    /**
     * 添加项目客户关联
     */
    int insert(ProjectCustomers projectCustomers);
    
    /**
     * 批量添加项目客户关联
     */
    int batchInsert(@Param("list") List<ProjectCustomers> list);
    
    /**
     * 更新项目客户关联
     */
    int update(ProjectCustomers projectCustomers);
    
    /**
     * 删除项目客户关联
     */
    int delete(@Param("id") String id);
    
    /**
     * 设置主客户（同时取消其他主客户）
     */
    int setPrimaryCustomer(@Param("projectId") String projectId, 
                          @Param("customerId") String customerId);
    
    /**
     * 统计项目的客户数量
     */
    int countByProjectId(@Param("projectId") String projectId);
    
    /**
     * 批量查询项目的客户数量
     */
    List<java.util.Map<String, Object>> countByProjectIds(@Param("projectIds") List<String> projectIds);
}
```

### 3.4 创建 Mapper XML

```xml
<?xml version="1.0" encoding="UTF-8" ?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" 
"http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.ruoyi.web.mapper.ProjectCustomersMapper">
    
    <resultMap type="ProjectCustomers" id="ProjectCustomersResult">
        <result property="id" column="id"/>
        <result property="projectId" column="project_id"/>
        <result property="customerId" column="customer_id"/>
        <result property="role" column="role"/>
        <result property="isPrimary" column="is_primary"/>
        <result property="joinDate" column="join_date"/>
        <result property="leaveDate" column="leave_date"/>
        <result property="isActive" column="is_active"/>
        <result property="shareRatio" column="share_ratio"/>
        <result property="remarks" column="remarks"/>
        <result property="createdAt" column="created_at"/>
        <result property="updatedAt" column="updated_at"/>
        <result property="createdBy" column="created_by"/>
        <result property="updatedBy" column="updated_by"/>
    </resultMap>
    
    <resultMap type="ProjectCustomers" id="ProjectCustomersWithCustomerResult" 
               extends="ProjectCustomersResult">
        <association property="customer" javaType="Customers">
            <id property="id" column="customer_id"/>
            <result property="name" column="customer_name"/>
            <result property="phone" column="customer_phone"/>
            <result property="email" column="customer_email"/>
            <result property="avatar" column="customer_avatar"/>
        </association>
    </resultMap>
    
    <!-- 查询项目的所有客户 -->
    <select id="selectByProjectId" resultMap="ProjectCustomersWithCustomerResult">
        SELECT 
            pc.id, pc.project_id, pc.customer_id, pc.role, pc.is_primary,
            pc.join_date, pc.leave_date, pc.is_active, pc.share_ratio, pc.remarks,
            pc.created_at, pc.updated_at, pc.created_by, pc.updated_by,
            c.name as customer_name, c.phone as customer_phone,
            c.email as customer_email, c.avatar as customer_avatar
        FROM project_customers pc
        LEFT JOIN customers c ON pc.customer_id = c.id
        WHERE pc.project_id = #{projectId} AND pc.is_active = 1
        ORDER BY pc.is_primary DESC, pc.join_date ASC
    </select>
    
    <!-- 查询客户的所有项目 -->
    <select id="selectByCustomerId" resultMap="ProjectCustomersResult">
        SELECT *
        FROM project_customers
        WHERE customer_id = #{customerId} AND is_active = 1
        ORDER BY is_primary DESC, join_date DESC
    </select>
    
    <!-- 查询项目的主客户 -->
    <select id="selectPrimaryByProjectId" resultMap="ProjectCustomersWithCustomerResult">
        SELECT 
            pc.id, pc.project_id, pc.customer_id, pc.role, pc.is_primary,
            pc.join_date, pc.leave_date, pc.is_active, pc.share_ratio, pc.remarks,
            c.name as customer_name, c.phone as customer_phone,
            c.email as customer_email, c.avatar as customer_avatar
        FROM project_customers pc
        LEFT JOIN customers c ON pc.customer_id = c.id
        WHERE pc.project_id = #{projectId} 
          AND pc.is_primary = 1 
          AND pc.is_active = 1
        LIMIT 1
    </select>
    
    <!-- 检查客户是否关联到项目 -->
    <select id="checkCustomerInProject" resultType="boolean">
        SELECT COUNT(*) > 0
        FROM project_customers
        WHERE project_id = #{projectId} 
          AND customer_id = #{customerId}
          AND is_active = 1
    </select>
    
    <!-- 添加项目客户关联 -->
    <insert id="insert" parameterType="ProjectCustomers">
        INSERT INTO project_customers (
            id, project_id, customer_id, role, is_primary,
            join_date, is_active, share_ratio, remarks,
            created_at, created_by
        ) VALUES (
            #{id}, #{projectId}, #{customerId}, #{role}, #{isPrimary},
            #{joinDate}, #{isActive}, #{shareRatio}, #{remarks},
            #{createdAt}, #{createdBy}
        )
    </insert>
    
    <!-- 批量添加 -->
    <insert id="batchInsert">
        INSERT INTO project_customers (
            id, project_id, customer_id, role, is_primary,
            join_date, is_active, share_ratio, remarks,
            created_at, created_by
        ) VALUES
        <foreach collection="list" item="item" separator=",">
            (
                #{item.id}, #{item.projectId}, #{item.customerId}, 
                #{item.role}, #{item.isPrimary}, #{item.joinDate}, 
                #{item.isActive}, #{item.shareRatio}, #{item.remarks},
                #{item.createdAt}, #{item.createdBy}
            )
        </foreach>
    </insert>
    
    <!-- 更新 -->
    <update id="update" parameterType="ProjectCustomers">
        UPDATE project_customers
        <set>
            <if test="role != null">role = #{role},</if>
            <if test="isPrimary != null">is_primary = #{isPrimary},</if>
            <if test="leaveDate != null">leave_date = #{leaveDate},</if>
            <if test="isActive != null">is_active = #{isActive},</if>
            <if test="shareRatio != null">share_ratio = #{shareRatio},</if>
            <if test="remarks != null">remarks = #{remarks},</if>
            <if test="updatedAt != null">updated_at = #{updatedAt},</if>
            <if test="updatedBy != null">updated_by = #{updatedBy},</if>
        </set>
        WHERE id = #{id}
    </update>
    
    <!-- 删除 -->
    <delete id="delete">
        DELETE FROM project_customers WHERE id = #{id}
    </delete>
    
    <!-- 设置主客户 -->
    <update id="setPrimaryCustomer">
        <!-- 先取消该项目的所有主客户标记 -->
        UPDATE project_customers 
        SET is_primary = 0, updated_at = NOW()
        WHERE project_id = #{projectId};
        
        <!-- 再设置新的主客户 -->
        UPDATE project_customers
        SET is_primary = 1, updated_at = NOW()
        WHERE project_id = #{projectId} AND customer_id = #{customerId};
    </update>
    
    <!-- 统计项目的客户数量 -->
    <select id="countByProjectId" resultType="int">
        SELECT COUNT(*)
        FROM project_customers
        WHERE project_id = #{projectId} AND is_active = 1
    </select>
    
    <!-- 批量统计 -->
    <select id="countByProjectIds" resultType="java.util.Map">
        SELECT 
            project_id as projectId,
            COUNT(*) as customerCount
        FROM project_customers
        WHERE project_id IN
        <foreach collection="projectIds" item="id" open="(" separator="," close=")">
            #{id}
        </foreach>
        AND is_active = 1
        GROUP BY project_id
    </select>
</mapper>
```
